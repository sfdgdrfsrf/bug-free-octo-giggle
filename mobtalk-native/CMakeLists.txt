cmake_minimum_required(VERSION 3.21)
project(MobTalkNative VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
FetchContent_Declare(preloader_android GIT_REPOSITORY https://github.com/LiteLDev/preloader-android.git GIT_TAG 0.2.3)
FetchContent_MakeAvailable(preloader_android)
add_library(mobtalk_native SHARED src/main.cpp src/mobtalk.cpp)
target_include_directories(mobtalk_native PRIVATE include)
target_link_libraries(mobtalk_native PRIVATE preloader log)
target_compile_options(mobtalk_native PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(mobtalk_native PROPERTIES OUTPUT_NAME "mobtalk_native" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
