cmake_minimum_required(VERSION 3.21)
project(LeviManiaMobile VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
FetchContent_Declare(preloader_android GIT_REPOSITORY https://github.com/LiteLDev/preloader-android.git GIT_TAG 0.2.3)
FetchContent_MakeAvailable(preloader_android)
add_library(levimania_mobile SHARED src/main.cpp src/core/config.cpp src/features/feature_registry.cpp src/features/systems.cpp src/features/port_engines.cpp)
target_include_directories(levimania_mobile PRIVATE include)
target_link_libraries(levimania_mobile PRIVATE preloader log)
target_compile_options(levimania_mobile PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(levimania_mobile PROPERTIES OUTPUT_NAME "levimania_mobile" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)

add_library(mobtalk_native SHARED mobtalk-native/src/main.cpp mobtalk-native/src/mobtalk.cpp mobtalk-native/src/persistence.cpp mobtalk-native/src/ui.cpp mobtalk-native/src/runtime_bridge.cpp mobtalk-native/src/entity_snapshot.cpp mobtalk-native/src/entity_tracker.cpp mobtalk-native/src/entity_recognition.cpp)
target_include_directories(mobtalk_native PRIVATE mobtalk-native/include)
target_link_libraries(mobtalk_native PRIVATE preloader log dl)
target_compile_options(mobtalk_native PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(mobtalk_native PROPERTIES OUTPUT_NAME "mobtalk_native" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
